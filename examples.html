Examples

Small, canonical programs for the v2 core. Each script runs against the shipped config (config.example.yaml); set NLIR=/path/to/nlir to choose a binary.

script needs a model what it shows
structure.sh no the structural core — Functor/Foldable, Monoid append and power, composition, lens get, data trees — with every output asserted
meaning.sh yes the semantic core — => :: .. ! & Δ ▷ ~> ≈ — and the prelude forms built from it
exact-over-fuzzy.sh yes the model reads, the evaluator counts: fuzzy extraction under exact folds and filters
spec-roundtrip.sh no a rule spec loaded as data, projected to prose and parsed back exactly (the spec lab's library)
lens.sh no backward passes: edit the prose a program wrote and recover the spec field (or program literal) that would produce it, with the lens laws checked
gapfill.sh no gap-filled specs: per-field lists of the points a spec implies and the decisions it leaves open, and the list-guided trim that keeps only what states them — the branch logic and the trim prompt checked
replay.sh yes the realisation store: record once, replay deterministically, --replay strict audits

The design these follow is docs/design/v2.md; the full operator reference is nlir help or SPEC.md.

exact-over-fuzzy.sh {#exact-over-fuzzy.sh}

#!/usr/bin/env bash
# The model reads; the evaluator counts. Neither alone does this reliably.
set -euo pipefail
NLIR=${NLIR:-nlir}
show() { printf '\n$ %s\n' "$1"; "$NLIR" --quiet -e "$1"; }
show "{\$0+\$1} ⊘ ['3 apples', '5 oranges', '2 pears']"                     # fuzzy read, exact sum → 10
show "#(\$filter%({\$0 ~> 'a complaint'}, ['love the app', 'it keeps crashing', 'when is dark mode', 'checkout is totally broken']))"
show "urgent={\$0 ~> 'urgent'}; \$if%(#(\$filter%(\$urgent, ['server is down', 'lunch plans', 'payments failing'])) >= 2, 'page on-call', 'queue for morning')"

gapfill.sh {#gapfill.sh}

#!/usr/bin/env bash
# Gap-filled specs, offline: the deterministic skeleton of the spec lab's gap-filled writer.
# A spec implies more points than its authors' prose makes, and leaves some decisions open. Two short
# lists per field close the gap — named points (implied, singled out) and additions (not implied) —
# and a list-guided trim keeps only what states them. Every output here is checked; with a model
# (--mode llm) the trim prompt goes to the model instead of being echoed.
set -euo pipefail
cd "$(dirname "$0")/.."
NLIR=${NLIR:-nlir}
LISTS="p={author_topics={MET=['states the tests that exercise the change'], UNCERTAIN=[]}, author_additions={MET=['names the test file']}}"
TRIM="keep_listed={\$if%(\$gt%(#(\$get%(\$p.author_topics,\$0,[])) + #(\$get%(\$p.author_additions,\$0,[])), 0), =>('Keep only what states a listed point. POINTS: '++\$join%(\$get%(\$p.author_topics,\$0,[]),'; ')++' | ADDITIONS: '++\$join%(\$get%(\$p.author_additions,\$0,[]),'; ')++' | DRAFT: '++\$1), \$1)}"
run() { "$NLIR" --config config.example.yaml --mode det --quiet -e "$LISTS; $TRIM; $1"; }
check() {
  local got; got=$(run "$1")
  if [ "$got" != "$2" ]; then printf 'FAIL  %s\n  got:      %s\n  expected: %s\n' "$1" "$got" "$2"; exit 1; fi
  printf 'ok    %-58s → %s\n' "$1" "$got"
}
# How many points and additions the authors listed for a field (a field's lists may be empty).
check "#(\$get%(\$p.author_topics,'MET',[])) + #(\$get%(\$p.author_additions,'MET',[]))" "2"
check "#(\$get%(\$p.author_topics,'UNCERTAIN',[]))"                                          "0"
# A field with no listed points passes through untouched: the trim never rewrites it.
check "\$keep_listed%('UNCERTAIN','Abstain when the diff is generated code.')"            "Abstain when the diff is generated code."
check "\$keep_listed%('NOT MET','No change to behaviour.')"                              "No change to behaviour."
# A listed field goes to the model with exactly its lists and the draft (det mode echoes the prompt).
check "\$keep_listed%('MET','A long draft.')" "response: (Keep only what states a listed point. POINTS: states the tests that exercise the change | ADDITIONS: names the test file | DRAFT: A long draft.)"
echo "gapfill: all checks passed"

lens.sh {#lens.sh}

#!/usr/bin/env bash
# Backward passes (lenses), offline: edit the prose a program produced and recover
# the spec edit (or program edit) that would produce it. Every result is checked.
set -euo pipefail
cd "$(dirname "$0")/../site/spec-lab"
NLIR=${NLIR:-nlir}
work=$(mktemp -d "${TMPDIR:-/tmp}/nlir-lens.XXXXXX")
trap 'rm -rf "$work"' EXIT
prog='$render%(($node%"CR V 1.1").profile)'
run() { "$NLIR" --mode det --quiet --bind spec=toy-spec.json -f spec.nlir -f profile.nlir -e "$prog"; }
put() { "$NLIR" put --edited "$1" --bind spec=toy-spec.json -f spec.nlir -f profile.nlir "$prog"; }
field() { python3 -c "import json,sys; r=json.load(sys.stdin); e=r['edits']; assert r['put_get'] is True, r; assert len(e)==1, e; print(e[0]['kind'], e[0]['target'], e[0]['changes'][0]['path'] or e[0].get('ast_path'))"; }

run > "$work/out.md"
# 1. A content edit to the MET criterion → the exact spec field.
python3 - "$work/out.md" "$work/content.md" <<'PY'
import sys
lines = open(sys.argv[1]).read().split("\n")
i = lines.index("> MET:") + 1
lines[i] = lines[i].rstrip(".") + ", or on the strength of the author's assurance alone."
open(sys.argv[2], "w").write("\n".join(lines))
PY
got=$(put "$work/content.md" | field)
[ "$got" = "data spec nodes/CR V 1.1/profile/decision_boundary/1" ] || { echo "FAIL content edit: $got"; exit 1; }
echo "ok    prose edit to MET          → $got"
# 2. A template edit to a heading → the program literal that wrote it.
sed 's/#### `decision_boundary`/#### `decision boundary`/' "$work/out.md" > "$work/template.md"
got=$(put "$work/template.md" | field)
case "$got" in "program statement "*) ;; *) echo "FAIL template edit: $got"; exit 1;; esac
echo "ok    heading edit               → $got"
# 3. No edit → no edits (GetPut).
got=$(put "$work/out.md" | python3 -c "import json,sys; r=json.load(sys.stdin); print(len(r['edits']), r['get_put'], r['put_get'])")
[ "$got" = "0 True True" ] || { echo "FAIL get-put: $got"; exit 1; }
echo "ok    unedited output            → no edits (GetPut, PutGet hold)"
echo "lens: all checks passed"

meaning.sh {#meaning.sh}

#!/usr/bin/env bash
# The semantic core and the prelude (needs a configured model; outputs vary by model).
set -euo pipefail
NLIR=${NLIR:-nlir}
show() { printf '\n$ %s\n' "$1"; "$NLIR" --quiet -e "$1"; }
show "=>'name three prime numbers, comma-separated'"                    # obey
show "'lmk if any Qs' :: formal"                                        # recast
show "'the release slipped because QA found a migration bug' :: 'a haiku'"
show "'the planets from the sun' .. 3"                                  # project a position
show "'the deploy failed twice on a flaky integration test' .. 'the cause'"
show "!'the release is on track'"                                       # dual
show "'ship Friday' & 'keep the rollback ready'"                        # product
show "a='Ship on Friday.'; b='Ship on Friday, after the security review.'; [\$a Δ \$b, \$a ▷ (\$a Δ \$b)]"
show "'the server keeps crashing' ~> 'this is urgent'"                  # entails
show "'Ship on Friday.' ≈ 'We release at the end of the week.'"         # similarity
# The prelude: everyday rewrites as named forms over :: and ..
show "(\$gist ∘ \$formal) % 'hey so the thing is the cache keeps evicting stuff we still need'"
show "\$question ↦ ['the build is green', 'the migration is reversible']"

replay.sh {#replay.sh}

#!/usr/bin/env bash
# Deterministic replay: model calls are recorded by content address and replayed.
set -euo pipefail
NLIR=${NLIR:-nlir}
store=$(mktemp -t nlir-store.XXXXXX)
trap 'rm -f "$store"' EXIT
expr="=>'Invent a name for a cat that lives with a lighthouse keeper'"
time "$NLIR" --quiet --cache-file "$store" -e "$expr"      # recorded
time "$NLIR" --quiet --cache-file "$store" -e "$expr"      # replayed: identical, instant
"$NLIR" --quiet --cache-file "$store" --replay strict -e "$expr"
"$NLIR" --quiet --cache-file "$store" -e "_seed=1; $expr"  # a new seed draws a fresh sample
"$NLIR" --quiet --cache-file "$store" -e "_sep=' | '; \$draw%({$expr}, 3)"

spec-roundtrip.sh {#spec-roundtrip.sh}

#!/usr/bin/env bash
# A rule spec as native data: project each rule's profile to prose and parse it back, exactly.
set -euo pipefail
cd "$(dirname "$0")/../site/spec-lab"
NLIR=${NLIR:-nlir}
lab() { "$NLIR" --mode det --quiet --bind spec=toy-spec.json -f spec.nlir -f profile.nlir -e "$1"; }
echo "rules: $(lab '#$leaves')"
lab "\$render%((\$node%'CR V 2.2').profile)" | head -12
echo "…"
lab "ok={\$json%(\$fields%(\$render%((\$node%\$0).profile))) == \$json%((\$node%\$0).profile)}; #(\$filter%(\$ok, \$leaves)) ++ '/' ++ #\$leaves ++ ' rules round-trip exactly'"

structure.sh {#structure.sh}

#!/usr/bin/env bash
# The structural core, offline: every line is checked.
set -euo pipefail
cd "$(dirname "$0")/.."
NLIR=${NLIR:-nlir}
run() { "$NLIR" --config config.example.yaml --mode det --quiet -e "$1"; }
check() {
  local got; got=$(run "$1")
  if [ "$got" != "$2" ]; then printf 'FAIL  %s\n  got:      %s\n  expected: %s\n' "$1" "$got" "$2"; exit 1; fi
  printf 'ok    %-58s → %s\n' "$1" "$got"
}
check "_sep=',';{\$0*\$0} ↦ [1,2,3]"                         "1,4,9"           # Functor map
check "{\$0+\$1} ⊘ (1..100)"                                  "5050"            # Foldable fold
check "_sep=',';[1,2] ++ [3]"                                 "1,2,3"           # Monoid: lists
check "_sep=',';{a=1,b=2} ++ {b=3,c=4}"                       "a=1,b=3,c=4"     # Monoid: dicts, right-biased
check "++[a,b,c]"                                             "abc"             # mconcat
check "'ab' _ 3"                                              "ababab"          # Monoid power
check "({\$0+1} _ 3) % 5"                                     "8"               # a form composed with itself
check "inc={\$0+1}; dbl={\$0*2}; (\$dbl ∘ \$inc) % 5"            "12"              # composition
check "#[a,b,c] + #{k=v} + #'abcd'"                           "8"               # length, polymorphic
check "[Mercury,Venus,Earth,Mars].2"                          "Earth"           # lens get
check "orders=[{who=ada,total=12.5},{who=bo,total=40},{who=cy,total=7}];\$join%({\$0.who} ↦ (\$filter%({\$0.total>=10},\$orders)),', ')" "ada, bo"
check "\$if%(3 < 5, yes, no)"                                  "yes"
echo "structure: all checks passed"
